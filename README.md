-- Schema for ATHAA donation website
-- This file should be run in Supabase SQL editor.

create table if not exists public.ranges (
  id bigint generated by default as identity primary key,
  range_name text not null unique,
  created_at timestamptz default now()
);

create table if not exists public.donations (
  id bigint generated by default as identity primary key,
  name text not null,
  madrasa text not null,
  range text not null,
  amount integer not null check (amount > 0),
  upi text not null default '7025144934@ptyes',
  created_at timestamptz default now()
);

insert into public.ranges (range_name)
values
  ('കണ്ണപുരം'),
  ('മാടായി'),
  ('ചെറുകുന്ന്'),
  ('പുതിയങ്ങാടി'),
  ('മാട്ടൂൽ')
on conflict (range_name) do nothing;

alter table public.ranges enable row level security;
alter table public.donations enable row level security;

create policy "Allow public read access to ranges"
on public.ranges
for select
using (true);

create policy "Allow public read access to donations"
on public.donations
for select
using (true);

create policy "Allow public insert to donations"
on public.donations
for insert
with check (true);

create policy "Allow public insert to ranges"
on public.ranges
for insert
with check (true);

-- Optional: If you want admin-only management then replace the insert/select policies with auth.uid() checks.
